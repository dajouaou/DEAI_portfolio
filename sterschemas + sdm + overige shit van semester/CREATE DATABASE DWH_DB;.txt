CREATE DATABASE DWH_DB;

CREATE TABLE Leverancier(
leveranciernr INTEGER PRIMARY KEY,
naam TEXT,
adres TEXT,
woonplaats TEXT);

CREATE TABLE Fabrikant(
fabrikantnr INTEGER PRIMARY KEY,
naam TEXT,
adres TEXT,
plaats TEXT);

CREATE TABLE Product(
productnr INTEGER PRIMARY KEY,
product_type TEXT,
soort TEXT,
merk TEXT,
type TEXT,
kleur TEXT,
standaardprijs REAL,
verkoopprijs REAL,
leverancier TEXT,
fabrikant TEXT
FOREIGN KEY (leverancier) REFERENCES Leverancier(leveranciernr),
FOREIGN KEY (fabrikant) REFERENCES Fabrikant(fabrikantnr));

CREATE TABLE Monteur(
monteurnr INTEGER PRIMARY KEY,
naam TEXT,
woonplaats TEXT,
uurloon REAL);

CREATE TABLE Filiaal(
filiaalnr INTEGER PRIMARY KEY,
naam TEXT,
adres TEXT,
provincie TEXT);

CREATE TABLE Datum(
datum INTEGER,
dag INTEGER,
maand TEXT,
kwartaal TEXT,
jaar TEXT);

CREATE TABLE Onderhoud(
onderhoudnr INTEGER PRIMARY KEY,
datum DATE,
productnr INTEGER NOT NULL,
monteurnr INTEGER NOT NULL,
filiaalnr INTEGER,
starttijd TIME,
eindtijd TIME,
kosten REAL,
duur TIME
FOREIGN KEY(productnr) REFERENCES Product(productnr),
FOREIGN KEY(monteurnr) REFERENCES Monteur(monteurnr),
FOREIGN KEY(filiaalnr) REFERENCES Filiaal(filiaalnr));

CREATE TABLE Klant(
klantnr INTEGER PRIMARY KEY,
naam TEXT,
woonplaats TEXT,
adres TEXT,
geslacht CHAR(1),
geboortedatum DATE);

CREATE TABLE InkoopPeriode(
periodenr INTEGER PRIMARY KEY,
inkoopmaand TEXT,
inkoopjaar INTEGER);

CREATE TABLE Inkoop(
inkoopnr INTEGER PRIMARY KEY,
productnr INTEGER,
periodenr INTEGER,
leveranciernr INTEGER,
fabrikantnr INTEGER,
aantal INTEGER,
inkoopprijs REAL,
inkoopwaarde INTEGER)
FOREIGN KEY(productnr) REFERENCES Product(productnr),
FOREIGN KEY(periodenr) REFERENCES InkoopPeriode(periodenr),
FOREIGN KEY(leveranciernr) REFERENCES Leverancier(leveranciernr),
FOREIGN KEY(fabrikantnr) REFERENCES Fabrikant(fabrikantnr));

CREATE TABLE Verkoop(
verkoopnr INTEGER PRIMARY KEY,
productnr INTEGER,
klantnr INTEGER,
monteurnr INTEGER,
filiaalnr INTEGER,
verkoopprijs REAL,
aantal INTEGER,
omzet REAL
FOREIGN KEY(productnr) REFERENCES Product(productnr),
FOREIGN KEY(klantnr) REFERENCES Klant(klantnr),
FOREIGN KEY(monteurnr) REFERENCES Monteur(monteurnr),
FOREIGN KEY(filiaalnr) REFERENCES Filiaal(filiaalnr));
